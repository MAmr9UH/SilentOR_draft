import gurobipy as gp

def solve(data):
    model = gp.Model()
    
    # Define variables
    w_vars = {}
    for i in range(len(data['components'])):
        for j in range(len(data['spare_levels'])):
            for k in range(len(data['spare_levels'])):
                if len(data['components']) == 3:
                    var_name = f"w_{i+1}_{j}_{k}"
                    w_vars[var_name] = model.addVar(vtype=gp.GRB.BINARY, name=var_name)
                elif len(data['components']) == 2:
                    var_name = f"w_{i+1}_{j}"
                    w_vars[var_name] = model.addVar(vtype=gp.GRB.BINARY, name=var_name)
    
    # Define objective function
    obj_func = gp.quicksum(w_vars[f"w_{i+1}_{j}_{k}"] * data['unit_price'][str(i+1)] for i in range(len(data['components'])) 
                           for j in range(len(data['spare_levels'])) 
                           for k in range(len(data['spare_levels'])))
    model.setObjective(obj_func, gp.GRB.MINIMIZE)
    
    # Define constraints
    budget_constraint = gp.quicksum(w_vars[f"w_{i+1}_{j}_{k}"] * data['unit_price'][str(i+1)] 
                                    for i in range(len(data['components'])) 
                                    for j in range(len(data['spare_levels'])) 
                                    for k in range(len(data['spare_levels']))) <= data['budget']
    model.addConstr(budget_constraint, "budget")
    
    weight_limit_constraint = gp.quicksum(w_vars[f"w_{i+1}_{j}_{k}"] * data['unit_weight'][str(i+1)] 
                                          for i in range(len(data['components'])) 
                                          for j in range(len(data['spare_levels'])) 
                                          for k in range(len(data['spare_levels']))) <= data['weight_limit']
    model.addConstr(weight_limit_constraint, "weight")
    
    # Exactly one combination is selected
    select_one = gp.quicksum(w_vars[f"w_{i+1}_{j}_{k}"] 
                             for i in range(len(data['components'])) 
                             for j in range(len(data['spare_levels'])) 
                             for k in range(len(data['spare_levels']))) == 1
    model.addConstr(select_one, "select")
    
    # Solve the model
    model.optimize()
    
    # Get solution status and objective value
    status = model.Status
    obj_val = model.ObjVal
    
    # Get variable values
    var_values = {var_name: w_vars[var_name].X for var_name in w_vars}
    
    return {
        "status": status,
        "obj_val": obj_val,
        "variables": var_values
    }
