from gurobipy import GRB
import math

    # Initialize decision variables

                "Tue": model.addVar(name="h_1_Tue", lb=0),

                "Thu": model.addVar(name="h_1_Thu", lb=0),
                "Fri": model.addVar(name="h_1_Fri", lb=0)},
            2: {"Mon": model.addVar(name="h_2_Mon", lb=0),

                "Thu": model.addVar(name="h_2_Thu", lb=0),
                "Fri": model.addVar(name="h_2_Fri", lb=0)},
            3: {"Mon": model.addVar(name="h_3_Mon", lb=0),
                "Tue": model.addVar(name="h_3_Tue", lb=0),
                "Wed": model.addVar(name="h_3_Wed", lb=0),
                "Thu": model.addVar(name="h_3_Thu", lb=0),
                "Fri": model.addVar(name="h_3_Fri", lb=0)},

                "Thu": model.addVar(name="h_6_Thu", lb=0),
                "Fri": model.addVar(name="h_6_Fri", lb=0)}

            1: {"Mon": model.addVar(name="y_1_Mon", vtype=GRB.BINARY),

                "Tue": model.addVar(name="y_2_Tue", vtype=GRB.BINARY),
                "Wed": model.addVar(name="y_2_Wed", vtype=GRB.BINARY),
                "Thu": model.addVar(name="y_2_Thu", vtype=GRB.BINARY),
                "Fri": model.addVar(name="y_2_Fri", vtype=GRB.BINARY)},
            3: {"Mon": model.addVar(name="y_3_Mon", vtype=GRB.BINARY),
                "Tue": model.addVar(name="y_3_Tue", vtype=GRB.BINARY),
                "Wed": model.addVar(name="y_3_Wed", vtype=GRB.BINARY),
                "Thu": model.addVar(name="y_3_Thu", vtype=GRB.BINARY),
                "Fri": model.addVar(name="y_3_Fri", vtype=GRB.BINARY)},
            4: {"Mon": model.addVar(name="y_4_Mon", vtype=GRB.BINARY),
                "Tue": model.addVar(name="y_4_Tue", vtype=GRB.BINARY),
                "Wed": model.addVar(name="y_4_Wed", vtype=GRB.BINARY),
                "Thu": model.addVar(name="y_4_Thu", vtype=GRB.BINARY),
                "Fri": model.addVar(name="y_4_Fri", vtype=GRB.BINARY)},
            5: {"Mon": model.addVar(name="y_5_Mon", vtype=GRB.BINARY),
                "Tue": model.addVar(name="y_5_Tue", vtype=GRB.BINARY),
                "Wed": model.addVar(name="y_5_Wed", vtype=GRB.BINARY),
                "Thu": model.addVar(name="y_5_Thu", vtype=GRB.BINARY),
                "Fri": model.addVar(name="y_5_Fri", vtype=GRB.BINARY)},
            6: {"Mon": model.addVar(name="y_6_Mon", vtype=GRB.BINARY),
                "Tue": model.addVar(name="y_6_Tue", vtype=GRB.BINARY),
                "Wed": model.addVar(name="y_6_Wed", vtype=GRB.BINARY),
                "Thu": model.addVar(name="y_6_Thu", vtype=GRB.BINARY),
                "Fri": model.addVar(name="y_6_Fri", vtype=GRB.BINARY)}
        }
    }
    
    # Objective function: Minimize gross pay
    wage = data["wage"]
    model.setObjective(
        gp.quicksum(wage[str(student)] * variables["h"][student][day] for student in range(1, 7) for day in ["Mon", "Tue", "Wed", "Thu", "Fri"]),
        GRB.MINIMIZE)

    # Each day, exactly one student is on duty
    for day in ["Mon", "Tue", "Wed", "Thu", "Fri"]:
        model.addConstr(gp.quicksum(variables["y"][student][day] for student in range(1, 7)) == 1)

    # Each student can work no more than 2 shifts per week
    for student in range(1, 7):
        model.addConstr(gp.quicksum(variables["y"][student][day] for day in ["Mon", "Tue", "Wed", "Thu", "Fri"]) <= 2)

    # Each student can work no more than max_students_per_day students on a given day
    for day in ["Mon", "Tue", "Wed", "Thu", "Fri"]:
        model.addConstr(gp.quicksum(variables["y"][student][day] for student in range(1, 7)) <= 3)

    # Each undergraduate must work at least 8 hours per week
    for student in [1, 2, 3, 4]:
        model.addConstr(variables["h"][student]["Mon"] + variables["h"][student]["Tue"] + variables["h"][student]["Wed"] + variables["h"][student]["Thu"] + variables["h"][student]["Fri"] >= 8)

    # Each graduate must work at least 7 hours per week
    for student in [5, 6]:
        model.addConstr(variables["h"][student]["Mon"] + variables["h"][student]["Tue"] + variables["h"][student]["Wed"] + variables["h"][student]["Thu"] + variables["h"][student]["Fri"] >= 7)

    # Availability hours
    availability_hours = data["availability_hours"]
    for student in range(1, 7):
        for day in ["Mon", "Tue", "Wed", "Thu", "Fri"]:
            model.addConstr(variables["h"][student][day] <= availability_hours[str(student)][day])

    return model, variables
