import gurobipy as gp
from gurobipy import GRB
import math

def build_model(data: dict) -> tuple:
    model = gp.Model("container_transportation_model")
    
    warehouses = data["warehouses"]
    ports = data["ports"]
    supply = data["supply"]
    demand = data["demand"]
    distance_km = data["distance_km"]
    truck_capacity_containers = data["truck_capacity_containers"]
    cost_per_km_per_truck = data["cost_per_km_per_truck"]

    # Decision variables
    x = {}
    for warehouse in warehouses:
        for port in ports:
            x[warehouse, port] = model.addVar(name=f"x_{warehouse}_{port}", vtype=GRB.INTEGER, lb=0)

    t = {}
    for warehouse in warehouses:
        for port in ports:
            t[warehouse, port] = model.addVar(name=f"t_{warehouse}_{port}", vtype=GRB.INTEGER, lb=0)

    # Objective function: Minimize total transportation cost
    model.setObjective(
        gp.quicksum(cost_per_km_per_truck * distance_km[warehouse][port] * x[warehouse, port] for warehouse in warehouses for port in ports),
        GRB.MINIMIZE)

    # Supply constraint
    for warehouse in warehouses:
        model.addConstr(gp.quicksum(x[warehouse, port] for port in ports) <= supply[warehouse])

    # Demand constraint
    for port in ports:
        model.addConstr(gp.quicksum(x[warehouse, port] for warehouse in warehouses) >= demand[port])

    # Each truck can carry up to 2 containers
    for warehouse in warehouses:
        for port in ports:
            model.addConstr(x[warehouse, port] <= 2 * t[warehouse, port])

    return model, {"x": x, "t": t}
