import gurobipy as gp
from gurobipy import GRB
import math

def build_model(data: dict) -> tuple:
    model = gp.Model("container_packing_model")
    
    # Decision variables
    y = {}
    uA = {}
    q = {}

    for i in data["container_ids"]:
        y[i] = model.addVar(name=f"y_{i}", vtype=GRB.BINARY)
        uA[i] = model.addVar(name=f"uA_{i}", vtype=GRB.BINARY)
        for good in data["goods"]:
            q[f"q_{i}_{good}"] = model.addVar(name=f"q_{i}_{good}", vtype=GRB.INTEGER, lb=0)

    # Objective function: Minimize the number of used containers
    model.setObjective(gp.quicksum(y[i] for i in data["container_ids"]), GRB.MINIMIZE)

    # Container capacity constraint
    for i in data["container_ids"]:
        model.addConstr(
            0.5 * q[f"q_{i}_A"] + 1.0 * q[f"q_{i}_B"] + 0.4 * q[f"q_{i}_C"] + 0.6 * q[f"q_{i}_D"] + 0.65 * q[f"q_{i}_E"] <= 60 * y[i]
        )

    # Each used container must load at least 18 tons
    for i in data["container_ids"]:
        model.addConstr(
            0.5 * q[f"q_{i}_A"] + 1.0 * q[f"q_{i}_B"] + 0.4 * q[f"q_{i}_C"] + 0.6 * q[f"q_{i}_D"] + 0.65 * q[f"q_{i}_E"] >= 18 * y[i]
        )

    # Each used container must load at least 12 units of D
    for i in data["container_ids"]:
        model.addConstr(q[f"q_{i}_D"] >= 12 * y[i])

    # If any A goods are loaded in a container, at least one unit of C must also be loaded in that container
    for i in data["container_ids"]:
        model.addConstr(q[f"q_{i}_A"] <= 1000 * uA[i])
        model.addConstr(q[f"q_{i}_C"] >= q[f"q_{i}_A"])

    # Total quantity of each good to be loaded
    for good in data["goods"]:
        model.addConstr(
            q[f"q_1_{good}"] + q[f"q_2_{good}"] + q[f"q_3_{good}"] + q[f"q_4_{good}"] + q[f"q_5_{good}"] + q[f"q_6_{good}"] + q[f"q_7_{good}"] + q[f"q_8_{good}"] + q[f"q_9_{good}"] + q[f"q_10_{good}"] == data["quantity"][good]
        )

    # Total weight of goods in each container
    for i in data["container_ids"]:
        model.addConstr(
            0.5 * q[f"q_{i}_A"] + 1.0 * q[f"q_{i}_B"] + 0.4 * q[f"q_{i}_C"] + 0.6 * q[f"q_{i}_D"] + 0.65 * q[f"q_{i}_E"] <= 60 * y[i]
        )
