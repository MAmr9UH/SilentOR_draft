import gurobipy as gp
from gurobipy import GRB
import math

def build_model(data: dict) -> tuple:
    model = gp.Model("warehouse_rental_model")

    demand = data["demand_100sqm"]
    contract_lengths = data["contract_lengths"]
    fee_per_100sqm_by_length = data["fee_per_100sqm_by_length"]
    feasible_start_length_pairs = data["feasible_start_length_pairs"]
    min_distinct_lengths = data["min_distinct_lengths"]
    max_distinct_lengths = data["max_distinct_lengths"]
    mutually_exclusive_lengths = data["mutually_exclusive_lengths"]

    months = data["months"]
    number_of_months = len(months)

    variables_keys = {
        "x_1_1": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_1_2": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_1_3": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_1_4": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_2_1": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_2_2": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_2_3": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_3_1": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_3_2": "integer number of 100-square-meter contracts starting in month s with length l",
        "x_4_1": "integer number of 100-square-meter contracts starting in month s with length l",
        "y_1": "binary 1 if any contract of length l is used",
        "y_2": "binary 1 if any contract of length l is used",
        "y_3": "binary 1 if any contract of length l is used",
        "y_4": "binary 1 if any contract of length l is used"
    }

    # Initialize decision variables
    x = {}
    for start_month in months:
        for length in contract_lengths:
            x[f"x_{start_month}_{length}"] = model.addVar(name=variables_keys[f"x_{start_month}_{length}"], vtype=GRB.INTEGER, lb=0)

    y = {}
    for length in contract_lengths:
        y[f"y_{length}"] = model.addVar(name=variables_keys[f"y_{length}"], vtype=GRB.BINARY)

    # Objective function: Minimize total rental cost
    total_rental_cost = 0
    for start_month in months:
        for length in contract_lengths:
            total_rental_cost += x[f"x_{start_month}_{length}"] * fee_per_100sqm_by_length[str(length)] * length

    model.setObjective(total_rental_cost, GRB.MINIMIZE)

    # Demand constraint
    for month in months:
        demand_in_100sqm = demand[str(month)]
        model.addConstr(gp.quicksum(x[f"x_{month}_{length}"] * length for length in contract_lengths if [month, length] in feasible_start_length_pairs) >= demand_in_100sqm)

    # At least two different contract lengths
    model.addConstr(gp.quicksum(y[f"y_{length}"] for length in contract_lengths) >= min_distinct_lengths)

    # At most three different contract lengths
    model.addConstr(gp.quicksum(y[f"y_{length}"] for length in contract_lengths) <= max_distinct_lengths)

    # If 4-month contract is chosen, no 1-month contract is chosen
    model.addConstr(y["y_4"] + y["y_1"] <= 1)

    # Solve the model
    return model, x, y

def solve(data: dict) -> dict:
    model, x, y = build_model(data)
    model.optimize()

    if model.status == GRB.OPTIMAL:
        solution = {
            "status": "OPTIMAL",
            "objective": model.objVal,
            "solution": {
                "x_1_1": x["x_1_1"].x,
                "x_1_2": x["x_1_2"].x,
                "x_1_3": x["x_1_3"].x,
                "x_1_4": x["x_1_4"].x,
                "x_2_1": x["x_2_1"].x,
                "x_2_2": x["x_2_2"].x,
                "x_2_3": x["x_2_3"].x,
                "x_3_1": x["x_3_1"].x,
                "x_3_2": x["x_3_2"].x,
                "x_4_1": x["x_4_1"].x,
                "y_1": y["y_1"].x,
                "y_2": y["y_2"].x,
                "y_3": y["y_3"].x,
                "y_4": y["y_4"].x
            }
        }
    else:
        solution = {
            "status": "INFEASIBLE",
            "objective": None,
            "solution": {
                "x_1_1": None,
                "x_1_2": None,
                "x_1_3": None,
                "x_1_4": None,
                "x_2_1": None,
                "x_2_2": None,
                "x_2_3": None,
                "x_3_1": None,
                "x_3_2": None,
                "x_4_1": None,
                "y_1": None,
                "y_2": None,
                "y_3": None,
                "y_4": None
            }
        }

    return solution
