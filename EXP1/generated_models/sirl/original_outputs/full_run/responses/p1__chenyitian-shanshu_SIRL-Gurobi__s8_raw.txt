import gurobipy as gp
from gurobipy import GRB
import math

def build_model(data: dict) -> tuple:
    model = gp.Model("plastic_production_model")
    
    cap = data["cap"]
    dem = data["dem"]
    vcost = data["vcost"]
    fixed = data["fixed"]
    
    # Decision variables
    produced = {}
    for i in range(1, 7):
        produced[i] = model.addVar(name=f"produced_{i}", lb=0, vtype=GRB.CONTINUOUS)
    
    allocation = {}
    for i in range(1, 7):
        for j in range(1, 7):
            if cap[i-1] >= cap[j-1]:
                allocation[(i, j)] = model.addVar(name=f"allocation_{i}_{j}", lb=0, vtype=GRB.CONTINUOUS)
    
    # Objective function: Minimize total cost
    model.setObjective(
        gp.quicksum(vcost[i-1] * produced[i] for i in range(1, 7)) +
        gp.quicksum(fixed * allocation[(i, j)] for i in range(1, 7) for j in range(1, 7) if cap[i-1] >= cap[j-1]),
        GRB.MINIMIZE)
    
    # Demand constraint
    for j in range(1, 7):
        model.addConstr(gp.quicksum(produced[i] for i in range(1, 7)) >= dem[j-1] - gp.quicksum(allocation[(i, j)] for i in range(1, 7) if cap[i-1] >= cap[j-1]))
    
    return model, {"produced": produced, "allocation": allocation}

def solve(data: dict) -> dict:
    model, variables = build_model(data)
    model.optimize()
    
    if model.status == GRB.OPTIMAL:
        solution = {
            "status": "OPTIMAL",
            "objective": model.objVal,
            "solution": {
                "produced": {i: variables["produced"][i].x for i in range(1, 7)},
                "allocation": { (i, j): variables["allocation"][(i, j)].x for i in range(1, 7) for j in range(1, 7) if data["cap"][i-1] >= data["cap"][j-1] }
            }
        }
    else:
        solution = {
            "status": "INFEASIBLE",
            "objective": None,
            "solution": {
                "produced": {},
                "allocation": {}
            }
        }
    
    return solution
