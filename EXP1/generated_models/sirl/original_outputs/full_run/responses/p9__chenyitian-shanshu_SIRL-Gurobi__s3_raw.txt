import gurobipy as gp
from gurobipy import GRB
import math

def build_model(data: dict) -> tuple:
    model = gp.Model("post_office_hiring_model")
    
    # Initialize decision variables
    variables = {}
    for day in range(7):
        variables[f"s{day}"] = model.addVar(name=f"s{day}", vtype=GRB.INTEGER, lb=0)

    # Employees needed on each day
    employees_needed = {
        "Mon": 15,
        "Tue": 13,
        "Wed": 15,
        "Thu": 18,
        "Fri": 14,
        "Sat": 16,
        "Sun": 10
    }

    # Build constraints
    for day in range(7):
        model.addConstr(
            variables["s0"] * (day == 0) +
            variables["s1"] * (day == 1) +
            variables["s2"] * (day == 2) +
            variables["s3"] * (day == 3) +
            variables["s4"] * (day == 4) +
            variables["s5"] * (day == 5) +
            variables["s6"] * (day == 6) ==
            employees_needed[list(employees_needed.keys())[day]]
        )

        if day < 5:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s0"] + variables["s1"] + variables["s2"] + variables["s3"] + variables["s4"] + variables["s5"] + variables["s6"]
            )

        if day > 0:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s1"] + variables["s2"] + variables["s3"] + variables["s4"] + variables["s5"] + variables["s6"] + variables["s0"]
            )

        if day > 1:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s2"] + variables["s3"] + variables["s4"] + variables["s5"] + variables["s6"] + variables["s0"] + variables["s1"]
            )

        if day > 2:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s3"] + variables["s4"] + variables["s5"] + variables["s6"] + variables["s0"] + variables["s1"] + variables["s2"]
            )

        if day > 3:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s4"] + variables["s5"] + variables["s6"] + variables["s0"] + variables["s1"] + variables["s2"] + variables["s3"]
            )

        if day > 4:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s5"] + variables["s6"] + variables["s0"] + variables["s1"] + variables["s2"] + variables["s3"] + variables["s4"]
            )

        if day > 5:
            model.addConstr(
                variables["s0"] +
                variables["s1"] +
                variables["s2"] +
                variables["s3"] +
                variables["s4"] +
                variables["s5"] +
                variables["s6"] >=
                variables["s6"] + variables["s0"] + variables["s1"] + variables["s2"] + variables["s3"] + variables["s4"] + variables["s5"]
            )

    # Objective function: Minimize total number of employees
    model.setObjective(
        variables["s0"] +
        variables["s1"] +
        variables["s2"] +
        variables["s3"] +
        variables["s4"] +
        variables["s5"] +
        variables["s6"],
        GRB.MINIMIZE)

    return model, variables

def solve(data: dict) -> dict:
    model, variables = build_model(data)
    model.optimize()

    if model.status == GRB.OPTIMAL:
        solution = {
            "s0": variables["s0"].x,
            "s1": variables["s1"].x,
            "s2": variables["s2"].x,
            "s3": variables["s3"].x,
            "s4": variables["s4"].x,
            "s5": variables["s5"].x,
            "s6": variables["s6"].x
        }
        return {
            "status": "OPTIMAL",
            "objective": model.objVal,
            "solution": solution
        }
    else:
        return {
            "status": "INFEASIBLE",
            "objective": None,
            "solution": None
        }
