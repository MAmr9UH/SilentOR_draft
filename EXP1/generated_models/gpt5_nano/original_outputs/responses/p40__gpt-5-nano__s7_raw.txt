import gurobipy as gp
from gurobipy import GRB

def build_model(data: dict) -> tuple:
    model = gp.Model()

    # Retrieve data
    max_skilled = data["max_skilled"]
    max_labor = data["max_labor"]
    ratio_max = data["skilled_to_labor_ratio_max"]
    task_hours = data["task_effective_hours"]
    req = data["method_worker_requirements"]
    wk_skill = data["weekly_wage"]["skilled"]
    wk_labor = data["weekly_wage"]["labor"]
    fixed_task1_B = data["method_worker_requirements"]["task1_B"].get("fixed_setup_cost", 0)

    # Decision variables (binary method choices)
    y1_A = model.addVar(vtype=GRB.BINARY, name="y1_A")
    y1_B = model.addVar(vtype=GRB.BINARY, name="y1_B")
    y2_A = model.addVar(vtype=GRB.BINARY, name="y2_A")
    y2_B = model.addVar(vtype=GRB.BINARY, name="y2_B")
    y3_A = model.addVar(vtype=GRB.BINARY, name="y3_A")
    y3_B = model.addVar(vtype=GRB.BINARY, name="y3_B")

    # Worker count variables per task/method (continuous to allow fractional values)
    skilled_t1_A = model.addVar(vtype=GRB.CONTINUOUS, name="skilled_t1_A")
    skilled_t1_B = model.addVar(vtype=GRB.CONTINUOUS, name="skilled_t1_B")
    skilled_t2_A = model.addVar(vtype=GRB.CONTINUOUS, name="skilled_t2_A")
    skilled_t3_B = model.addVar(vtype=GRB.CONTINUOUS, name="skilled_t3_B")

    labor_t1_B = model.addVar(vtype=GRB.CONTINUOUS, name="labor_t1_B")
    labor_t2_B = model.addVar(vtype=GRB.CONTINUOUS, name="labor_t2_B")
    labor_t3_A = model.addVar(vtype=GRB.CONTINUOUS, name="labor_t3_A")
    labor_t3_B = model.addVar(vtype=GRB.CONTINUOUS, name="labor_t3_B")

    total_skilled = model.addVar(vtype=GRB.CONTINUOUS, name="total_skilled")
    total_labor = model.addVar(vtype=GRB.CONTINUOUS, name="total_labor")

    # Objective: minimize weekly wages + fixed costs
    objective = wk_skill * total_skilled + wk_labor * total_labor + fixed_task1_B * y1_B
    model.setObjective(objective, GRB.MINIMIZE)

    # Linking equations: only one method per task
    model.addConstr(y1_A + y1_B == 1, name="one_method_task1")
    model.addConstr(y2_A + y2_B == 1, name="one_method_task2")
    model.addConstr(y3_A + y3_B == 1, name="one_method_task3")

    # Exclusion: if task 1 uses B, task 3 cannot use pure-labor (A)
    model.addConstr(y1_B + y3_A <= 1, name="exclusion_task1B_task3A")

    # Minimum skilled if task 3 uses B
    model.addConstr(total_skilled - 20 * y3_B >= 0, name="min_skilled_if_task3_B")

    # Total skilled and labor from the per-method counts
    model.addConstr(total_skilled == skilled_t1_A + skilled_t1_B + skilled_t2_A + skilled_t3_B, name="sum_skilled")
    model.addConstr(total_labor == labor_t1_B + labor_t2_B + labor_t3_A + labor_t3_B, name="sum_labor")

    # Capacity limits
    model.addConstr(total_skilled <= max_skilled, name="max_skilled_cap")
    model.addConstr(total_labor <= max_labor, name="max_labor_cap")
    model.addConstr(total_skilled <= ratio_max * total_labor, name="ratio_skill_to_labor")

    # Equality constraints tying counts to method choices
    # Task 1: A uses 200 skilled; B uses 200 skilled and 400 labor
    skill_t1_A_target = req["task1_A"]["skilled"]
    labor_t1_B_target = req["task1_B"]["labor"]
    skill_t1_B_target = req["task1_B"]["skilled"]

    model.addConstr(skilled_t1_A <= skill_t1_A_target * y1_A, name="t1A_skill_upper")
    model.addConstr(skilled_t1_A >= skill_t1_A_target * y1_A, name="t1A_skill_lower")

    model.addConstr(skilled_t1_B <= skill_t1_B_target * y1_B, name="t1B_skill_upper")
    model.addConstr(skilled_t1_B >= skill_t1_B_target * y1_B, name="t1B_skill_lower")

    model.addConstr(labor_t1_B <= labor_t1_B_target * y1_B, name="t1B_labor_upper")
    model.addConstr(labor_t1_B >= labor_t1_B_target * y1_B, name="t1B_labor_lower")

    # Task 2: A uses 257.142857... skilled; B uses 300 labor
    skill_t2_A_target = req["task2_A"]["skilled"]
    labor_t2_B_target = req["task2_B"]["labor"]

    model.addConstr(skilled_t2_A <= skill_t2_A_target * y2_A, name="t2A_skill_upper")
    model.addConstr(skilled_t2_A >= skill_t2_A_target * y2_A, name="t2A_skill_lower")

    model.addConstr(labor_t2_B <= labor_t2_B_target * y2_B, name="t2B_labor_upper")
    model.addConstr(labor_t2_B >= labor_t2_B_target * y2_B, name="t2B_labor_lower")

    # Task 3: A uses 500 labor; B uses 120 skilled and 360 labor (groups of 1 skilled + 3 labor)
    skill_t3_B_target = req["task3_B"]["skilled"]
    labor_t3_A_target = req["task3_A"]["labor"]
    labor_t3_B_target = req["task3_B"]["labor"]

    model.addConstr(skilled_t3_B <= skill_t3_B_target * y3_B, name="t3B_skill_upper")
    model.addConstr(skilled_t3_B >= skill_t3_B_target * y3_B, name="t3B_skill_lower")

    model.addConstr(labor_t3_B <= labor_t3_B_target * y3_B, name="t3B_labor_upper")
    model.addConstr(labor_t3_B >= labor_t3_B_target * y3_B, name="t3B_labor_lower")

    model.addConstr(labor_t3_A <= labor_t3_A_target * y3_A, name="t3A_labor_upper")
    model.addConstr(labor_t3_A >= labor_t3_A_target * y3_A, name="t3A_labor_lower")

    # Hours constraints for each task
    model.addConstr(42 * skilled_t1_A + 42 * skilled_t1_B + 36 * labor_t1_B >= int(task_hours["1"]), name="hours_task1")
    model.addConstr(42 * skilled_t2_A + 36 * labor_t2_B >= int(task_hours["2"]), name="hours_task2")
    model.addConstr(42 * skilled_t3_B + 36 * labor_t3_B + 36 * labor_t3_A >= int(task_hours["3"]), name="hours_task3")

    # Expose variables to the caller
    variables = {
        "y1_A": y1_A,
        "y1_B": y1_B,
        "y2_A": y2_A,
        "y2_B": y2_B,
        "y3_A": y3_A,
        "y3_B": y3_B,
        "skilled_t1_A": skilled_t1_A,
        "skilled_t1_B": skilled_t1_B,
        "skilled_t2_A": skilled_t2_A,
        "skilled_t3_B": skilled_t3_B,
        "labor_t1_B": labor_t1_B,
        "labor_t2_B": labor_t2_B,
        "labor_t3_A": labor_t3_A,
        "labor_t3_B": labor_t3_B,
        "total_skilled": total_skilled,
        "total_labor": total_labor
    }

    model.update()
    return model, variables


def solve(data: dict) -> dict:
    model, variables = build_model(data)
    model.optimize()

    }
